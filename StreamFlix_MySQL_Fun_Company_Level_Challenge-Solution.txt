-- Display title, genre, rating, price, views, director, and production company.
-- Conditions: rating > 8, views > 500,000, price BETWEEN 100 AND 250.
-- Sort: views DESC, Limit: 5.
SELECT title, genre, rating, price, views, director, production_company 
FROM movies 
WHERE rating > 8 
  AND views > 500000 
  AND price BETWEEN 100 AND 250 
ORDER BY views DESC 
LIMIT 5;
